import subprocess
def generate_rsa_keys():
    subprocess.run([
        "openssl", "genpkey",
        "-algorithm", "RSA",
        "-pkeyopt", "rsa_keygen_bits:2048",
        "-out", "private.pem"
    ], check=True)

    subprocess.run([
        "openssl", "pkey",
        "-in", "private.pem",
        "-pubout",
        "-out", "public.pem"
    ], check=True)

    with open("public.pem", "r") as file:
        print(file.read())

    print("Keys generated successfully")

import hashlib
import time
class Block:
    def __init__(self, index, timestamp, data, previous_hash):
        self.index = index
        self.timestamp = timestamp
        self.data = data
        self.previous_hash = previous_hash
        self.hash = self.calculate_hash()
    def calculate_hash(self):
        block_data = (
            str(self.index)
            + str(self.timestamp)
            + str(self.data)
            + str(self.previous_hash)
        )

        return hashlib.sha256(block_data.encode()).hexdigest()
class Blockchain:
    def __init__(self):
        self.chain = [self.create_genesis_block()]

    def create_genesis_block(self):
        return Block(
            0,
            time.ctime(),
            "Genesis Block",
            "0"
        )

    def get_latest_block(self):
        return self.chain[-1]

    def add_block(self, new_data):
        previous_block = self.get_latest_block()

        new_block = Block(
            len(self.chain),
            time.ctime(),
            new_data,
            previous_block.hash
        )

        self.chain.append(new_block)
    def is_chain_valid(self):
        for index in range(1, len(self.chain)):
            current_block = self.chain[index]
            previous_block = self.chain[index - 1]

            if current_block.hash != current_block.calculate_hash():
                print("Block", index, "has been tampered!")
                return False

            if current_block.previous_hash != previous_block.hash:
                print("Previous hash does not match!")
                return False

        return True

import cmd
from web3 import Web3, EthereumTesterProvider


class Wallet(cmd.Cmd):
    prompt = "(crypto)> "

    def __init__(self):
        super().__init__()
        self.w3 = Web3(EthereumTesterProvider())

        accounts = self.w3.eth.accounts
        self.users = {
            "bank": accounts[0],
            "alice": accounts[1],
            "bob": accounts[2],
            "charlie": accounts[3]
        }

        print("Blockchain started")
        print("Users: bank, alice, bob, charlie")
        print("Commands: balance, send, chain, exit")

    def do_balance(self, name):
        """Usage: balance alice"""

        name = name.lower().strip()

        if name not in self.users:
            print("User not found")
            return

        address = self.users[name]
        balance = self.w3.eth.get_balance(address)

        print("Address:", address)
        print("Balance:", self.w3.from_wei(balance, "ether"), "ETH")

    def do_send(self, text):
        """Usage: send bank alice 10"""

        try:
            sender, receiver, amount = text.lower().split()

            transaction = {
                "from": self.users[sender],
                "to": self.users[receiver],
                "value": self.w3.to_wei(amount, "ether")
            }

            tx_hash = self.w3.eth.send_transaction(transaction)
            receipt = self.w3.eth.wait_for_transaction_receipt(tx_hash)

            print("Transaction successful")
            print("Hash:", tx_hash.hex())
            print("Block:", receipt.blockNumber)
            print("Gas used:", receipt.gasUsed)

        except Exception as error:
            print("Transaction failed:", error)

    def do_chain(self, text):
        """Display the latest block."""

        block = self.w3.eth.get_block("latest")

        print("Block number:", block.number)
        print("Block hash:", block.hash.hex())
        print("Transactions:", len(block.transactions))

    def do_exit(self, text):
        """Close the program."""

        print("Closing wallet")
        return True


Wallet().cmdloop()
